# Use an official lightweight Node.js image
FROM node:20-slim

# Set the working directory inside the container
WORKDIR /app

# Copy dependency definition files first to leverage caching
COPY package*.json ./

# Install project dependencies
RUN npm install

# Copy the rest of your application source code
COPY . .

# Expose the port your app runs on (change 3000 to match your app)
EXPOSE 3000

# Specify the command to start your application
CMD ["npm", "start"]
